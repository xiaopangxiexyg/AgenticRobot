#!/usr/bin/env zsh
set -euo pipefail
umask 077

node_root=/share/zc/proxy/tools/node/v24.16.0
npm_prefix=/share/zc/proxy/tools/npm-global
npm_cache=/share/zc/proxy/npm-cache
npm_home=/share/zc/proxy/npm-home
backup_root=/share/zc/proxy/backups
package_dir=/share/zc/proxy/tools/npm-global/lib/node_modules/@openai/codex
package_json="$package_dir/package.json"
node="$node_root/bin/node"
npm="$node_root/bin/npm"

[[ -x "$node" ]]
[[ -x "$npm" ]]
[[ -f "$package_json" ]]
[[ -x /share/zc/codex ]]

install -d -m 700 "$npm_cache" "$npm_home"
mkdir -p "$backup_root"

export HOME="$npm_home"
export PATH="$node_root/bin:/usr/bin:/bin"
export NPM_CONFIG_PREFIX="$npm_prefix"
export NPM_CONFIG_CACHE="$npm_cache"
export HTTP_PROXY=http://10.8.2.248:17890
export HTTPS_PROXY="$HTTP_PROXY"
export http_proxy="$HTTP_PROXY"
export https_proxy="$HTTP_PROXY"
export NO_PROXY=127.0.0.1,localhost,::1
export no_proxy="$NO_PROXY"

current_version=$("$node" -e 'const p=require(process.argv[1]); process.stdout.write(p.version)' "$package_json")
latest_version=$("$npm" view @openai/codex version)
stamp=$(date -u +%Y%m%dT%H%M%SZ)
backup="$backup_root/codex-package-$current_version-$stamp"

printf 'current=%s\n' "$current_version"
printf 'latest=%s\n' "$latest_version"
printf 'backup=%s\n' "$backup"

[[ ! -e "$backup" ]]
cp -a "$package_dir" "$backup"

"$npm" install --global --prefix "$npm_prefix" --no-audit --no-fund "@openai/codex@$latest_version"

installed_version=$("$node" -e 'const p=require(process.argv[1]); process.stdout.write(p.version)' "$package_json")
[[ "$installed_version" == "$latest_version" ]]
[[ "$(readlink -f "$npm_prefix/bin/codex")" == "$package_dir/bin/codex.js" ]]

/share/zc/codex --version
printf 'updated=%s\n' "$installed_version"
printf 'rollback_version=%s\n' "$current_version"
printf 'backup=%s\n' "$backup"
