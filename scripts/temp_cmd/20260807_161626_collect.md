#!/usr/bin/env bash
set -Eeo pipefail

DEXTELEOP_ROOT=/home/user/zc/DexTeleop
CONDA_SH=/home/user/miniconda3/etc/profile.d/conda.sh
CONDA_ENV=/home/user/miniconda3/envs/dexteleop310
PYTHON_BIN="$CONDA_ENV/bin/python"
DATASET_ROOT=/home/user/zc/datasets/

for required_directory in \
  "$DEXTELEOP_ROOT" \
  "$CONDA_ENV" \
  "$DATASET_ROOT"; do
  if [[ ! -d "$required_directory" ]]; then
    printf '[DexTeleop] required directory not found: %s\n' "$required_directory" >&2
    exit 1
  fi
done

for required_file in \
  "$CONDA_SH" \
  "$DEXTELEOP_ROOT/docs/real/teleop_args.sh" \
  "$DEXTELEOP_ROOT/hardware/rm65/utils/configs/calibrations/rm65_right_DMTAC_M.json" \
  "$DEXTELEOP_ROOT/hardware/rm65/utils/configs/calibrations/rm65_left_DMTAC_M.json" \
  "$DEXTELEOP_ROOT/teleop_runtime/configs/gripper_task_profile.example.json" \
  "$DEXTELEOP_ROOT/teleop_runtime/configs/devices/realsense.json" \
  "$DEXTELEOP_ROOT/teleop_runtime/configs/hardware/q6/latest.json" \
  "$DEXTELEOP_ROOT/teleop_runtime/configs/hardware/dmtac/latest.json"; do
  if [[ ! -r "$required_file" ]]; then
    printf '[DexTeleop] required readable file not found: %s\n' "$required_file" >&2
    exit 1
  fi
done

for required_executable in /usr/bin/taskset "$PYTHON_BIN"; do
  if [[ ! -x "$required_executable" ]]; then
    printf '[DexTeleop] required executable not found: %s\n' "$required_executable" >&2
    exit 1
  fi
done

cd -- "$DEXTELEOP_ROOT"
export PYTHONPATH="$DEXTELEOP_ROOT${PYTHONPATH:+:$PYTHONPATH}"
export LEFT_ARM_IP=192.168.101.21
export RIGHT_ARM_IP=192.168.1.20
export DATASET_BASE="$DATASET_ROOT"

source "$CONDA_SH"
conda activate "$CONDA_ENV"
set -u
source docs/real/teleop_args.sh

COMMON_REAL_TELEOP_ARGS=(
  --active-sides both
  --right-arm-ip 192.168.1.20
  --left-arm-ip 192.168.101.21
  --right-tool-profile "hardware/rm65/utils/configs/calibrations/rm65_right_DMTAC_M.json"
  --left-tool-profile "hardware/rm65/utils/configs/calibrations/rm65_left_DMTAC_M.json"
  --gui
  --hand-backend dmtac-gripper
  --gripper-task-profile "teleop_runtime/configs/gripper_task_profile.example.json"
  --q6-hold-recalibration-gesture
  --no-episode-recording
  --arm-io-threads 4
)

DMTAC_CAPTURE_COMMON_ARGS=(
  --dmtac-with-tactile
  --dmtac-tactile-fields raw
  --dmtac-tactile-compute-profile requested
  --dmtac-tactile-read-hz 15
  --dmtac-tactile-process-read
  --dmtac-tactile-worker-cpus 12,13,14,15
  --dmtac-tactile-worker-nice 15
  --dmtac-tactile-worker-queue-size 4
  --dmtac-tactile-parallel-read
  --dmtac-tactile-no-denoise
  --dmtac-tactile-no-flat-vector
  --external-tactile
)

DMTAC_A_ARGS=(
  --dmtac-tactile-global-read-lock
)

DMTAC_TACTILE_WORKER_2CPU_ARGS=(
  --dmtac-tactile-worker-cpus 12,13
)

DATASET_COMMON_ARGS=(
  --no-session-record-commands
  --session-record-flush-every 60
  --session-record-compact-json
  --writer-workers 6
  --writer-queue-size 1024
  --storage-queue-size 128
  --wrist-depth-mask
)

POSITION_FORMAL_V2_BALANCED_ARGS=(
  --position-control-profile rm65_bimanual_position_v2_balanced
  --arm-control-mode position
  --position-ik-step-mode single_step_singularity_aware
  --no-position-ik-branch-continuity-anchor
  --position-joint-limit-guard-mode legacy
  --position-wrist-branch-switch-guard-deg 12.0
  --position-wrist-q5-hysteresis-release-deg 6.0
  --position-force-guard
  --position-force-guard-source tool
  --position-force-guard-max-force-n 30
  --position-force-guard-max-torque-nm 5
  --position-force-guard-hz 20
  --position-force-guard-consecutive-samples 1
  --position-force-guard-max-read-errors 3
  --position-force-guard-stale-sec 0.4
)

POSITION_COLLECTION_PROCESS_ARGS=(
  --collection-worker-mode process
  --collection-process-queue-size 4
  --collection-process-shared-slots 4
  --collection-process-extra-array-mb 8
  --collection-worker-cpus 8,9,10,11
  --collection-worker-nice 5
)

POSITION_COLLECTION_PRODUCTION_ARGS=(
  --control-debug off
  --no-debug-action
  --no-debug-timing-auto-log
  --actuator-trace
  --actuator-trace-root "teleop_records/actuator_identification"
  --actuator-trace-max-duration-sec 600
)

TELEOP_ARGS=(
  "${COMMON_REAL_TELEOP_ARGS[@]}"
  "${DMTAC_CAPTURE_COMMON_ARGS[@]}"
  "${DMTAC_A_ARGS[@]}"
  "${DMTAC_TACTILE_WORKER_2CPU_ARGS[@]}"
  "${DATASET_COMMON_ARGS[@]}"
  "${POSITION_FORMAL_V2_BALANCED_ARGS[@]}"
  "${POSITION_COLLECTION_PROCESS_ARGS[@]}"
  "${POSITION_COLLECTION_PRODUCTION_ARGS[@]}"
  --dataset-root "$DATASET_ROOT"
  --collection-task-name "dmtac_rm65_bimanual_position_v2_writing_4_stage"
  --collection-task-description "formal rm65 bimanual position v2 balanced"
  --q6-config latest
  --q6-config-required
  --dmtac-config latest
  --require-dmtac-config
  --camera-config "teleop_runtime/configs/devices/realsense.json"
  --camera-color-format bgr8
  --write-parquet
  --write-video
  --collection-progress-gui
  --gui-resizable-sidebars
  --no-tactile-artifact-integrity
  --idle-tactile-compression
)

position_profile_count=0
for index in "${!TELEOP_ARGS[@]}"; do
  argument=${TELEOP_ARGS[$index]}
  if [[ -z "${argument//[[:space:]]/}" ]]; then
    printf '[DexTeleop] refusing whitespace-only argument at index=%s value=%q\n' \
      "$index" "$argument" >&2
    exit 2
  fi
  if [[ "$argument" == --position-control-profile ]]; then
    position_profile_count=$((position_profile_count + 1))
  fi
done

if [[ "$position_profile_count" -ne 1 ]]; then
  printf '[DexTeleop] expected exactly one --position-control-profile, got %s\n' \
    "$position_profile_count" >&2
  exit 2
fi

set +e
/usr/bin/taskset -c 0-7 \
  "$PYTHON_BIN" -m teleop_runtime.debug.run_vive_rm65_real_teleop \
  "${TELEOP_ARGS[@]}"
rc=$?
set -e

printf '[DexTeleop] command exited with status=%s\n' "$rc"
exit "$rc"
