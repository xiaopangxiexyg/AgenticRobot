(
set -Eeuo pipefail

swap_path=/swapfile2
swap_size=6G
fstab_line='/swapfile2 none swap sw 0 0'
minimum_free_bytes=$((12 * 1024 * 1024 * 1024))
available_bytes=$(df --output=avail -B1 / | tail -n 1 | tr -d ' ')

free -h
df -h /
swapon --show

if (( available_bytes < minimum_free_bytes )); then
  printf 'Refusing to add %s: less than 12 GiB is available on /.\n' "$swap_size" >&2
  exit 2
fi

if [[ -e "$swap_path" ]]; then
  if ! swapon --noheadings --raw --output NAME | grep -Fqx "$swap_path"; then
    printf 'Refusing to overwrite existing inactive path: %s\n' "$swap_path" >&2
    exit 2
  fi
else
  sudo fallocate -l "$swap_size" "$swap_path"
  sudo chmod 600 "$swap_path"
  sudo mkswap "$swap_path"
  sudo swapon "$swap_path"
fi

if ! grep -Fqx "$fstab_line" /etc/fstab; then
  sudo cp --preserve=all --backup=numbered /etc/fstab /etc/fstab.before-swapfile2
  printf '%s\n' "$fstab_line" | sudo tee -a /etc/fstab >/dev/null
fi

free -h
df -h /
swapon --show
)
