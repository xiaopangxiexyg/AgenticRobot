#!/usr/bin/env bash

(
set -Eeuo pipefail

project=/home/user/zc/DexTeleop
python_bin=/home/user/zc/openpi/.venv/bin/python
profile=deployments/openpi/configs/training/writing_stage4_0804_0805_front_h16_absolute_v1.json

cd "$project"
export CUDA_VISIBLE_DEVICES=0
export WANDB_MODE=offline

printf '[stage4] preparing and verifying the dataset view\n'
"$python_bin" -u -m deployments.openpi.training.prepared_skill_launch prepare \
  --profile "$profile"

printf '[stage4] checking OpenPI samples\n'
"$python_bin" -u -m deployments.openpi.training.prepared_skill_launch sample-check \
  --profile "$profile" \
  --skip-source-file-verification

printf '[stage4] computing normalization statistics\n'
"$python_bin" -u -m deployments.openpi.training.prepared_skill_launch norm-stats \
  --profile "$profile" \
  --stats-batch-size 1 \
  --stats-num-workers 0 \
  --skip-source-file-verification

printf '[stage4] training; press S to save and continue, or Ctrl+C once to save and exit\n'
exec "$python_bin" -u -m deployments.openpi.training.prepared_skill_launch train \
  --profile "$profile"
)
